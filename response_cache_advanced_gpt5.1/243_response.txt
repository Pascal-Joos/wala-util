      throws IllegalArgumentException, UnimplementedError {

      MutableIntSet pCopy = makeMutableCopy(db.primaryImpl);
      MutableIntSet sCopy = makeMutableCopy(db.secondaryImpl);

      assert pCopy.sameValue(db.primaryImpl);
