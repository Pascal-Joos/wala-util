      if (start < 0) {
        throw new IllegalArgumentException("illegal start: " + start);
      }
      if (bits == null) {
        return -1;
      }
      int word = subscript(start);

      while (word < bits.length) {
        if (bits[word] != 0) {
          do {
            if ((bits[word] & bit) != 0) return start;
            bit <<= 1;
            start++;
          } while (bit != 0);
        } else {

        }

        bit = 1;
      }

  }
